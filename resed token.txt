    // FLUJO COMPLETO: GENERAR PDF, SUBIR A STORAGE, REGISTRAR Y NOTIFICAR
    // =========================================================================
    async function procesarYFinalizarInspeccion() {
      const btn = document.getElementById('btn_guardar_cloud');
      const originalHTML = btn.innerHTML;
      btn.innerHTML = `<i class="fa-solid fa-spinner fa-spin mr-1"></i> Generando PDF...`;
      btn.disabled = true;

      try {
        if (!ticketId || !supabaseClient) {
          throw new Error("No hay un Ticket ID válido en la sesión actual.");
        }

        // 1. Obtener datos del ticket de Supabase
        const { data: ticketData, error: ticketErr } = await supabaseClient
          .from('tickets')
          .select('*')
          .eq('id', ticketId)
          .single();

        if (ticketErr || !ticketData) {
          throw new Error("No se pudo leer el ticket de Supabase.");
        }

        const nombreCliente = ticketData.cliente || "Cliente General";
        const codigoInforme = `PRO-TECH-2026-${Math.random().toString(36).substring(2, 8).toUpperCase()}`;

        // 2. Asegurar renderizado del modal para capturarlo correctamente con html2pdf
        const modal = document.getElementById('modal_dossier_informe');
        const estabaOculto = modal.classList.contains('hidden');
        if (estabaOculto) {
          await abrirModalInforme();
          await new Promise(resolve => setTimeout(resolve, 800));
        }

        const elementoInforme = document.getElementById('dossier_contenido_body');
        if (!elementoInforme) {
          throw new Error("No se encontró el contenedor del informe para generar el PDF.");
        }

        const opcionesPdf = {
          margin:       10,
          filename:     `Dossier_${codigoInforme}.pdf`,
          image:        { type: 'jpeg', quality: 0.98 },
          html2canvas:  { scale: 2, useCORS: true, logging: false },
          jsPDF:        { unit: 'mm', format: 'a4', orientation: 'portrait' }
        };

        btn.innerHTML = `<i class="fa-solid fa-spinner fa-spin mr-1"></i> Convirtiendo PDF...`;
        
        // Generar archivo binario PDF mediante blob
        const pdfBlob = await html2pdf().from(elementoInforme).set(opcionesPdf).outputPdf('blob');
        
        if (estabaOculto) {
          cerrarModalInforme();
        }

        if (!pdfBlob) {
          throw new Error("Falló la conversión del documento a PDF.");
        }

        // 3. Subir al Bucket 'dossieres' de Supabase Storage
        btn.innerHTML = `<i class="fa-solid fa-spinner fa-spin mr-1"></i> Subiendo a Storage...`;
        const nombreArchivo = `${codigoInforme}.pdf`;
        
        const { error: uploadErr } = await supabaseClient.storage
          .from('dossieres')
          .upload(nombreArchivo, pdfBlob, {
            contentType: 'application/pdf',
            upsert: true
          });

        if (uploadErr) {
          throw new Error("Error al subir al Storage: " + uploadErr.message);
        }

        // Obtener URL pública del PDF recién subido
        const { data: urlData } = supabaseClient.storage
          .from('dossieres')
          .getPublicUrl(nombreArchivo);
        
        const urlPublicaPdf = urlData?.publicUrl || null;

        // 4. Registrar en la tabla dossieres_generados
        const { data: dossierInsertado, error: dossierError } = await supabaseClient
          .from('dossieres_generados')
          .insert([{
            ticket_id: ticketId,
            codigo_unico: codigoInforme,
            cliente_nombre: nombreCliente,
            enlace_pdf: urlPublicaPdf,
            resumen_datos: { ticket: ticketData },
            enviado_correo: false
          }])
          .select()
          .single();

        if (dossierError) {
          console.warn("Aviso registrando en dossieres_generados:", dossierError.message);
        }

        // 5. Enviar correo a través de la Edge Function pasando el urlPdf
        btn.innerHTML = `<i class="fa-solid fa-spinner fa-spin mr-1"></i> Enviando Correo...`;
        let correoOK = false;
        
        try {
          const { error: emailErr } = await supabaseClient.functions.invoke('enviar-correo', {
            body: {
              cliente: nombreCliente,
              codigoInforme: codigoInforme,
              correoDestino: "jpluna@protechsecurityec.com",
              urlPdf: urlPublicaPdf
            }
          });

          if (!emailErr) {
            correoOK = true;
            if (dossierInsertado?.id) {
              await supabaseClient
                .from('dossieres_generados')
                .update({ enviado_correo: true })
                .eq('id', dossierInsertado.id);
            }
          } else {
            console.error("Fallo reportado por Edge Function:", emailErr);
          }
        } catch (eMail) {
          console.error("Excepción en llamada de correo:", eMail);
        }

        // 6. Actualizar el ticket a completado
        await supabaseClient
          .from('tickets')
          .update({ estado: 'completado' })
          .eq('id', ticketId);

        alert(`¡Inspección finalizada exitosamente!\n\n` +
              `• Código Único: ${codigoInforme}\n` +
              `• PDF en Cloud: ${urlPublicaPdf ? 'GUARDADO' : 'NO DISPONIBLE'}\n` +
              `• Notificación con enlace: ${correoOK ? 'ENVIADA' : 'FALLÓ'}\n` +
              `• Ticket: COMPLETADO`);

        window.location.href = 'agenda_tickets.html';

      } catch (err) {
        console.error("Error al procesar la inspección:", err);
        alert("Ocurrió un error: " + err.message);
      } finally {
        btn.innerHTML = originalHTML;
        btn.disabled = false;
      }
    }
